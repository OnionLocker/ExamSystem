### ~/.hermes/config.yaml

```yaml
tool_loop_guardrails:
  hard_stop_enabled: true          # 改为 true（原 false）
  hard_stop_after:
    exact_failure: 3               # 收紧到 3（原 5）
    same_tool_failure: 4           # 收紧到 4（原 8）
    idempotent_no_progress: 3      # 收紧到 3（原 5）
agent:
  max_turns: 30                    # 降到 30（原 90）
```

### ~/.hermes/agent-hooks/pre-write.sh

```bash
#!/bin/bash
FILE="$1"

# 白名单
if [[ "$FILE" =~ ^data/(practice-reviews|exam-reviews|voice-notes|manual-) ]] || \
   [[ "$FILE" =~ ^hermes-skills/ ]] || \
   [[ "$FILE" =~ /.hermes/memories/ ]]; then
  exit 0
fi

# 禁区
if [[ "$FILE" =~ ^(src|server|scripts)/ ]] || \
   [[ "$FILE" =~ \.(lock|json)$ ]]; then
  echo "❌ 禁止写入 $FILE，需先说明并确认"
  exit 1
fi

exit 0
```

**安装步骤：**
```bash
chmod +x ~/.hermes/agent-hooks/pre-write.sh
```
